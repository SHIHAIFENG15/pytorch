### 验收标准

对于各种特性验收策略，NPU和GPU上执行**不小于1000 个step**，精度满足

 

MoE模型：

与基线数据相比：预训练 平均绝对误差<=0.02，或平均相对误差<=2%

非MoE模型：

与基线数据相比：预训练 平均绝对误差<=0.01，或平均相对误差<=1%

### 验收方法

日志中可直接拿到吞吐量数据 tps=tokens/sec，torchtian打印的tps为单卡吞吐量，并不是整个集群的总吞吐量

拿到这个数值后，按照以下步骤计算：

\1. 跑4卡基准：

  \- 设置 NGPU=4，运行训练，记录稳定后的 tps。

\2. 跑多卡测试：

  \- 设置 NGPU=8，运行训练，记录稳定后的 tps。

收集tps的脚本cal_tps.py

执行命令：python cal_tps.py xxx.log

```
#!/usr/bin/env python3
import argparse
import re
import sys

def calculate_robust_stats(values, skip_steps, trim_percent, precision, metric_name):
    """计算单一指标的稳健统计量"""
    total_found = len(values)
    if total_found == 0:
        return None

    # 1. 跳过 Warmup
    if skip_steps >= total_found:
        effective_data = values
    else:
        effective_data = values[skip_steps:]
        
    if not effective_data:
        return None

    # 2. 去除极值 (Trimmed Mean)
    n = len(effective_data)
    trim_count = int(n * (trim_percent / 100))
    
    if trim_count > 0 and n > 2 * trim_count:
        sorted_data = sorted(effective_data)
        trimmed_data = sorted_data[trim_count : n - trim_count]
        removed_info = f" (已去除首尾各 {trim_count} 个极值)"
    else:
        trimmed_data = effective_data
        removed_info = " (数据量过少，未进行极值去除)"

    if not trimmed_data:
        return None

    # 3. 计算结果
    final_avg = sum(trimmed_data) / len(trimmed_data)
    min_val = min(trimmed_data)
    max_val = max(trimmed_data)
    fmt_str = f".{precision}f"
    unit = "%" if metric_name == "MFU" else ""

    return {
        "name": metric_name,
        "total": total_found,
        "effective": len(effective_data),
        "final_count": len(trimmed_data),
        "removed_info": removed_info,
        "min": min_val,
        "max": max_val,
        "avg": final_avg,
        "fmt": fmt_str,
        "unit": unit
    }

def main():
    parser = argparse.ArgumentParser(
        description="从训练日志文件中提取指定指标并计算稳健平均值 (支持跳过 Warmup 和去除极值)",
        formatter_class=argparse.RawTextHelpFormatter
    )
    parser.add_argument("logfile", help="日志文件路径 (例如: train.log)")
    parser.add_argument("--skip-steps", type=int, default=100,
                        help="跳过前 N 个数据点 (即 Warmup 阶段，默认: 100)")
    parser.add_argument("--trim-percent", type=float, default=0.0,
                        help="去除首尾各百分之多少的极值 (0-50，默认: 0.0)")
    parser.add_argument("-p", "--precision", type=int, default=2,
                        help="输出平均值的小数位数 (默认: 2)")
    parser.add_argument("--metric", type=str, default="both", choices=["tps", "mfu", "both"],
                        help="要计算的指标: tps, mfu, 或 both (默认: both)")

    args = parser.parse_args()

    if not (0 <= args.trim_percent < 50):
        print("❌ 错误: --trim-percent 必须在 0 到 50 之间")
        sys.exit(1)

    # 匹配规则：兼容千分位逗号(TPS) 和百分号(MFU)
    tps_pattern = re.compile(r'tps:\s*([\d,]+(?:\.\d+)?)')
    mfu_pattern = re.compile(r'mfu:\s*([\d.]+)\s*%?')

    all_tps = []
    all_mfu = []

    try:
        with open(args.logfile, "r", encoding="utf-8", errors="ignore") as f:
            for line in f:
                tps_m = tps_pattern.search(line)
                if tps_m:
                    val = float(tps_m.group(1).replace(",", ""))
                    if val > 0:
                        all_tps.append(val)

                mfu_m = mfu_pattern.search(line)
                if mfu_m:
                    val = float(mfu_m.group(1))
                    if val > 0:
                        all_mfu.append(val)
    except FileNotFoundError:
        print(f"❌ 错误: 找不到文件 '{args.logfile}'")
        sys.exit(1)
    except Exception as e:
        print(f"❌ 读取文件时发生异常: {e}")
        sys.exit(1)

    # 根据 --metric 决定计算哪些指标
    targets = []
    if args.metric in ("tps", "both"):
        targets.append(("TPS", all_tps))
    if args.metric in ("mfu", "both"):
        targets.append(("MFU", all_mfu))

    print(f"📊 数据统计报告 (计算指标: {args.metric.upper()})")
    any_found = False
    
    for name, data in targets:
        stats = calculate_robust_stats(data, args.skip_steps, args.trim_percent, args.precision, name)
        if stats is None:
            print(f"⚠️  未找到有效的 {name} 数据或过滤后为空。")
            continue

        any_found = True
        print(f"--- {name} ---")
        print(f"   - 原始数据总数: {stats['total']}")
        print(f"   - 跳过前 {args.skip_steps} 步后剩余: {stats['effective']}")
        print(f"   - 最终参与计算的数据量: {stats['final_count']}{stats['removed_info']}")
        print(f"   - 最小值: {stats['min']:{stats['fmt']}}{stats['unit']} | 最大值: {stats['max']:{stats['fmt']}}{stats['unit']}")
        print(f"✅ {name} 稳健平均值: {stats['avg']:{stats['fmt']}}{stats['unit']}")
        print()

    if not any_found:
        print("⚠️  未在日志中找到任何有效数据。")
        sys.exit(0)

if __name__ == "__main__":
    main()
```

profiler.py如下，放在trainer.py同级目录下

```
import torch

# 尝试导入 torch_npu，用于判断是否支持 NPU
try:
    import torch_npu
    HAS_NPU = True
except ImportError:
    HAS_NPU = False

class COMMProfiler:
    def __init__(self, device='npu', enable=False, warmup=0, active=5, level="level2", 
                 with_stack=False, with_memory=True, record_shapes=True, save_path="./profile"):
        """
        通用 Profiler 类，支持 NPU 和 CUDA
        
        Args:
            device (str): 设备类型，'npu' 或 'cuda'，默认 'npu'
            enable (bool): 是否启用 profiler
            warmup (int): 预热步数
            active (int): 活跃记录步数
            level (str): NPU 特有，profiler 级别 (level0, level1, level2)，默认 'level2'
            with_stack (bool): 是否记录栈信息
            with_memory (bool): 是否记录内存信息
            record_shapes (bool): 是否记录算子形状
            save_path (str): 日志保存路径
        """
        self.enable = enable
        self.device = device.lower()
        
        # 公共参数
        self.warmup = warmup
        self.active = active
        self.sp_with_stack = with_stack
        self.sp_with_memory = with_memory
        self.sp_record_shapes = record_shapes
        self.sp_save_path = save_path
        
        # NPU 特有参数
        self.sp_level = level

        if not self.enable:
            self.prof = None
            return

        if self.device == 'cuda':
            self._init_cuda_profiler()
        elif self.device == 'npu':
            self._init_npu_profiler()
        else:
            raise ValueError(f"Unsupported device: {self.device}. Choose 'cuda' or 'npu'.")

    def _init_cuda_profiler(self):
        """初始化 CUDA Profiler"""
        activities = [
            torch.profiler.ProfilerActivity.CUDA,
            torch.profiler.ProfilerActivity.CPU
        ]
        
        self.prof = torch.profiler.profile(
            with_stack=self.sp_with_stack,
            record_shapes=self.sp_record_shapes,
            profile_memory=self.sp_with_memory,
            activities=activities,
            schedule=torch.profiler.schedule(wait=0, warmup=self.warmup, active=self.active, repeat=1),
            on_trace_ready=torch.profiler.tensorboard_trace_handler(self.sp_save_path)
        )

    def _init_npu_profiler(self):
        """初始化 NPU Profiler"""
        if not HAS_NPU:
            raise ImportError("torch_npu is not installed, but device='npu' was selected.")

        # 处理 Profiler Level
        if self.sp_level == 'level0':
            profiler_level = torch_npu.profiler.ProfilerLevel.Level0
        elif self.sp_level == 'level1':
            profiler_level = torch_npu.profiler.ProfilerLevel.Level1
        elif self.sp_level == 'level2':
            profiler_level = torch_npu.profiler.ProfilerLevel.Level2
        else:
            raise ValueError(f"profiler_level only supports level0, level1, and level2, but gets {self.sp_level}")
        
        # 如果开启 stack，强制 level2
        if self.sp_with_stack:
            profiler_level = torch_npu.profiler.ProfilerLevel.Level2

        # 配置 NPU 特有实验性参数
        experimental_config = torch_npu.profiler._ExperimentalConfig(
            export_type=[
                torch_npu.profiler.ExportType.Text,
                torch_npu.profiler.ExportType.Db
            ],
            profiler_level=profiler_level,
            msprof_tx=False,
            aic_metrics=torch_npu.profiler.AiCMetrics.AiCoreNone,
            l2_cache=False,
            op_attr=False,
            data_simplification=False,
            record_op_args=False,
            gc_detect_threshold=None
        )

        activities = [
            torch_npu.profiler.ProfilerActivity.NPU,
            torch_npu.profiler.ProfilerActivity.CPU
        ]

        self.prof = torch_npu.profiler.profile(
            with_stack=self.sp_with_stack,
            record_shapes=self.sp_record_shapes,
            profile_memory=self.sp_with_memory,
            activities=activities,
            schedule=torch_npu.profiler.schedule(wait=0, warmup=self.warmup, active=self.active, repeat=1),
            on_trace_ready=torch_npu.profiler.tensorboard_trace_handler(self.sp_save_path),
            experimental_config=experimental_config
        )

    def start(self):
        if self.enable and self.prof:
            self.prof.start()

    def step(self):
        if self.enable and self.prof:
            self.prof.step()

    def stop(self):
        if self.enable and self.prof:
            self.prof.stop()
```
